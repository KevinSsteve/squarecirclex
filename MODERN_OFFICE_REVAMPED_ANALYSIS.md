### 1. Arquitetura de Personagem Controlável

#### PlayerController System
```javascript
class PlayerController {
  constructor(scene, playerEntity) {
    this.scene = scene;
    this.player = playerEntity;
    this.speed = 100; // pixels per second
    this.inputEnabled = true;
  }
  
  update(deltaTime) {
    if (!this.inputEnabled) return;
    
    const input = this.getInput();
    const velocity = this.calculateVelocity(input);
    
    // Update player position
    this.player.getComponent('position').velocity = velocity;
    
    // Update animation based on movement
    this.updateAnimation(velocity);
  }
  
  getInput() {
    return {
      up: keyboard.isDown('W') || keyboard.isDown('ArrowUp'),
      down: keyboard.isDown('S') || keyboard.isDown('ArrowDown'),
      left: keyboard.isDown('A') || keyboard.isDown('ArrowLeft'),
      right: keyboard.isDown('D') || keyboard.isDown('ArrowRight')
    };
  }
  
  calculateVelocity(input) {
    let vx = 0, vy = 0;
    
    if (input.up) vy -= 1;
    if (input.down) vy += 1;
    if (input.left) vx -= 1;
    if (input.right) vx += 1;
    
    // Normalize diagonal movement
    if (vx !== 0 && vy !== 0) {
      const length = Math.sqrt(vx * vx + vy * vy);
      vx /= length;
      vy /= length;
    }
    
    return {
      x: vx * this.speed,
      y: vy * this.speed
    };
  }
  
  updateAnimation(velocity) {
    const animComp = this.player.getComponent('animation');
    
    if (velocity.x === 0 && velocity.y === 0) {
      animComp.play('idle');
    } else {
      animComp.play('walk');
      
      // Update direction
      const direction = this.getDirection(velocity);
      animComp.setDirection(direction);
    }
  }
  
  getDirection(velocity) {
    // 4 directions: up, down, left, right
    if (Math.abs(velocity.x) > Math.abs(velocity.y)) {
      return velocity.x > 0 ? 'right' : 'left';
    } else {
      return velocity.y > 0 ? 'down' : 'up';
    }
  }
}
```

### 2. Camera Follow System

```javascript
class CameraFollowSystem {
  constructor(scene, target) {
    this.scene = scene;
    this.target = target;
    this.smoothing = 0.1; // Lerp factor
    this.offset = { x: 0, y: 0 };
  }
  
  update(deltaTime) {
    const targetPos = this.target.getComponent('position');
    const camera = this.scene.camera;
    
    // Calculate desired camera position
    const desiredX = targetPos.x + this.offset.x;
    const desiredY = targetPos.y + this.offset.y;
    
    // Smooth lerp to target
    camera.x += (desiredX - camera.x) * this.smoothing;
    camera.y += (desiredY - camera.y) * this.smoothing;
    
    // Clamp to world bounds
    this.clampToBounds(camera);
  }
  
  clampToBounds(camera) {
    const bounds = this.scene.worldBounds;
    const viewport = this.scene.viewport;
    
    camera.x = Math.max(0, Math.min(camera.x, bounds.width - viewport.width));
    camera.y = Math.max(0, Math.min(camera.y, bounds.height - viewport.height));
  }
}
```

### 3. Collision System (Grid-Based)

```javascript
class CollisionSystem {
  constructor(scene) {
    this.scene = scene;
    this.grid = this.buildCollisionGrid();
    this.tileSize = 16;
  }
  
  buildCollisionGrid() {
    // Build grid from furniture and walls
    const grid = [];
    const furniture = this.scene.getFurniture();
    
    furniture.forEach(item => {
      const pos = item.getComponent('position');
      const gridX = Math.floor(pos.x / this.tileSize);
      const gridY = Math.floor(pos.y / this.tileSize);
      
      if (!grid[gridY]) grid[gridY] = [];
      grid[gridY][gridX] = true; // Blocked
    });
    
    return grid;
  }
  
  canMoveTo(x, y) {
    const gridX = Math.floor(x / this.tileSize);
    const gridY = Math.floor(y / this.tileSize);
    
    return !this.grid[gridY]?.[gridX];
  }
  
  resolveCollision(entity) {
    const pos = entity.getComponent('position');
    const vel = pos.velocity;
    
    // Check next position
    const nextX = pos.x + vel.x;
    const nextY = pos.y + vel.y;
    
    // Try X movement
    if (!this.canMoveTo(nextX, pos.y)) {
      vel.x = 0;
    }
    
    // Try Y movement
    if (!this.canMoveTo(pos.x, nextY)) {
      vel.y = 0;
    }
  }
}
```

### 4. Interaction System

```javascript
class InteractionSystem {
  constructor(scene, player) {
    this.scene = scene;
    this.player = player;
    this.interactionRange = 32; // pixels
    this.currentTarget = null;
  }
  
  update() {
    // Find nearest interactable
    const nearest = this.findNearestInteractable();
    
    // Update highlight
    if (nearest !== this.currentTarget) {
      if (this.currentTarget) {
        this.removeHighlight(this.currentTarget);
      }
      if (nearest) {
        this.addHighlight(nearest);
      }
      this.currentTarget = nearest;
    }
    
    // Handle interaction input
    if (keyboard.justPressed('E') && this.currentTarget) {
      this.interact(this.currentTarget);
    }
  }
  
  findNearestInteractable() {
    const playerPos = this.player.getComponent('position');
    const interactables = this.scene.getInteractables();
    
    let nearest = null;
    let minDist = this.interactionRange;
    
    interactables.forEach(entity => {
      const pos = entity.getComponent('position');
      const dist = this.distance(playerPos, pos);
      
      if (dist < minDist) {
        minDist = dist;
        nearest = entity;
      }
    });
    
    return nearest;
  }
  
  interact(entity) {
    const type = entity.type;
    
    switch (type) {
      case 'agent':
        this.showAgentDetails(entity);
        break;
      case 'desk':
        this.showDeskMenu(entity);
        break;
      case 'department':
        this.showHireMenu(entity);
        break;
    }
  }
  
  distance(pos1, pos2) {
    const dx = pos1.x - pos2.x;
    const dy = pos1.y - pos2.y;
    return Math.sqrt(dx * dx + dy * dy);
  }
}
```

### 5. Agent AI System (NPCs)

```javascript
class AgentAISystem {
  constructor(scene) {
    this.scene = scene;
    this.agents = [];
  }
  
  update(deltaTime) {
    this.agents.forEach(agent => {
      this.updateAgent(agent, deltaTime);
    });
  }
  
  updateAgent(agent, deltaTime) {
    const state = agent.getState();
    
    switch (state) {
      case 'working':
        this.handleWorking(agent, deltaTime);
        break;
      case 'idle':
        this.handleIdle(agent, deltaTime);
        break;
      case 'moving':
        this.handleMoving(agent, deltaTime);
        break;
    }
  }
  
  handleWorking(agent, deltaTime) {
    // Play working animation
    const animComp = agent.getComponent('animation');
    animComp.play('sit'); // Sitting at desk
    
    // Occasionally take a break
    agent.workTimer += deltaTime;
    if (agent.workTimer > 30) { // 30 seconds
      agent.workTimer = 0;
      this.goToCoffee(agent);
    }
  }
  
  handleIdle(agent, deltaTime) {
    // Play idle animation
    const animComp = agent.getComponent('animation');
    animComp.play('idle');
    
    // Return to work after a bit
    agent.idleTimer += deltaTime;
    if (agent.idleTimer > 5) {
      agent.idleTimer = 0;
      this.returnToDesk(agent);
    }
  }
  
  handleMoving(agent, deltaTime) {
    // Play walk animation
    const animComp = agent.getComponent('animation');
    animComp.play('walk');
    
    // Check if reached destination
    if (this.reachedDestination(agent)) {
      agent.setState('idle');
    }
  }
  
  goToCoffee(agent) {
    const coffeeMachine = this.scene.getCoffeeMachine();
    agent.setDestination(coffeeMachine.position);
    agent.setState('moving');
  }
  
  returnToDesk(agent) {
    const desk = agent.assignedDesk;
    agent.setDestination(desk.position);
    agent.setState('moving');
  }
}
```
